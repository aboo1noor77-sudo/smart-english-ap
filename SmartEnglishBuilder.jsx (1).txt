// ─── Claude API ───────────────────────────────────────────────────────────────
async function callClaude(prompt) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-20250514",
      max_tokens: 1000,
      messages: [{ role: "user", content: prompt }],
    }),
  });
  const data = await res.json();
  return data.content?.[0]?.text || "";
}

  const r = (size - stroke) / 2;
  const circ = 2 * Math.PI * r;

    <svg width={size} height={size} style={{ transform: "rotate(-90deg)" }}>

      <circle cx={size/2} cy={size/2} r={r} fill="none" stroke={color} strokeWidth={stroke}

// WELCOME
function WelcomeScreen({ onLogin }) {
  const [mode, setMode] = useState("main"); // main | login | register
  const [form, setForm] = useState({ name: "", email: "", pass: "" });
  const [err, setErr] = useState("");

  const handleLogin = () => {
    const users = storage.get("users") || {};
    if (!users[form.email] || users[form.email].pass !== form.pass) return setErr("البريد أو كلمة المرور غلط");
    onLogin(users[form.email]);
  };
  const handleRegister = () => {
    if (!form.name || !form.email || !form.pass) return setErr("أكمل جميع الحقول");
    const users = storage.get("users") || {};
    if (users[form.email]) return setErr("هذا البريد مسجل مسبقاً");
    const user = { name: form.name, email: form.email, pass: form.pass, joined: today() };
    users[form.email] = user;
    storage.set("users", users);
    onLogin(user);
  };

  return (
    <div style={{ minHeight: "100vh", display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center", padding: 24, background: `radial-gradient(ellipse at 60% 0%, ${theme.accent}0d 0%, transparent 60%), radial-gradient(ellipse at 20% 100%, ${theme.accent2}0d 0%, transparent 60%), ${theme.bg}` }}>
      {/* Logo */}
      <div className="fade-up" style={{ textAlign: "center", marginBottom: 48 }}>
        <div style={{ width: 80, height: 80, borderRadius: 24, background: `linear-gradient(135deg, ${theme.accent}, ${theme.accent2})`, display: "flex", alignItems: "center", justifyContent: "center", margin: "0 auto 20px", boxShadow: `0 0 40px ${theme.accent}44`, animation: "glow 3s ease infinite" }}>
          <Icon name="zap" size={36} />
        </div>
        <h1 style={{ fontFamily: G.fontDisplay, fontSize: 32, fontWeight: 800, letterSpacing: "-0.02em", marginBottom: 8 }}>
          Smart English<br /><span style={{ color: theme.accent }}>Builder</span>
        </h1>
        <p style={{ color: theme.muted, fontSize: 15, maxWidth: 280, margin: "0 auto" }}>
          تعلم كلمات جديدة كل يوم بطريقة ذكية
        </p>
      </div>

      {mode === "main" && (
        <div className="fade-up-2" style={{ width: "100%", maxWidth: 340, display: "flex", flexDirection: "column", gap: 12 }}>
          <button className="btn btn-primary btn-lg" style={{ width: "100%" }} onClick={() => setMode("login")}>
            <Icon name="arrow_right" size={18} /> تسجيل الدخول
          </button>
          <button className="btn btn-ghost btn-lg" style={{ width: "100%" }} onClick={() => setMode("register")}>
            <Icon name="plus" size={18} /> حساب جديد
          </button>
        </div>
      )}

      {(mode === "login" || mode === "register") && (
        <div className="fade-up-2 card" style={{ width: "100%", maxWidth: 360 }}>
          <h2 style={{ fontFamily: G.fontDisplay, fontSize: 20, fontWeight: 700, marginBottom: 20 }}>
            {mode === "login" ? "تسجيل الدخول" : "إنشاء حساب جديد"}
          </h2>
          <div style={{ display: "flex", flexDirection: "column", gap: 12 }}>
            {mode === "register" && (
              <input className="input" placeholder="الاسم" value={form.name} onChange={e => setForm(p => ({...p, name: e.target.value}))} />
            )}
            <input className="input" placeholder="البريد الإلكتروني" type="email" value={form.email} onChange={e => setForm(p => ({...p, email: e.target.value}))} />
            <input className="input" placeholder="كلمة المرور" type="password" value={form.pass} onChange={e => setForm(p => ({...p, pass: e.target.value}))} />
            {err && <p style={{ color: theme.danger, fontSize: 13 }}>{err}</p>}
            <button className="btn btn-primary" style={{ marginTop: 4 }} onClick={mode === "login" ? handleLogin : handleRegister}>
              {mode === "login" ? "دخول" : "إنشاء الحساب"}
            </button>
            <button className="btn btn-ghost" onClick={() => { setMode("main"); setErr(""); }}>رجوع</button>
          </div>
        </div>
      )}
    </div>
  );
}

// QUIZ COMPONENT
function Quiz({ questions, onFinish }) {
  const [qIdx, setQIdx] = useState(0);
  const [answers, setAnswers] = useState({});
  const [selected, setSelected] = useState(null);
  const [confirmed, setConfirmed] = useState(false);
  const [orderWords, setOrderWords] = useState([]);

  const q = questions[qIdx];

  useEffect(() => {
    setSelected(null); setConfirmed(false);
    if (q?.type === "order") setOrderWords([...q.words].sort(() => Math.random() - 0.5));
  }, [qIdx]);

  const confirm = () => {
    const ans = q.type === "order" ? orderWords.join(" ") : selected;
    setAnswers(p => ({...p, [qIdx]: ans}));
    setConfirmed(true);
  };

  const next = () => {
    if (qIdx < questions.length - 1) setQIdx(qIdx + 1);
    else onFinish(answers);
  };

  const isCorrect = () => {
    if (q.type === "truefalse") return answers[qIdx] === q.answer;
    if (q.type === "mcq") return answers[qIdx] === q.answer;
    if (q.type === "order") return answers[qIdx]?.toLowerCase().trim() === q.sentence?.toLowerCase().trim();
    return false;
  };

  return (
    <div className="fade-up" style={{ display: "flex", flexDirection: "column", gap: 16 }}>
      {/* Progress */}
      <div className="card" style={{ padding: 16 }}>
        <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 8 }}>
          <span style={{ fontSize: 13, color: theme.muted }}>السؤال {qIdx + 1} من {questions.length}</span>
          <span style={{ fontSize: 13, fontFamily: G.fontMono, color: theme.accent }}>{Math.round((qIdx / questions.length) * 100)}%</span>
        </div>
        <div style={{ height: 4, background: theme.border, borderRadius: 2 }}>
          <div style={{ height: "100%", width: `${((qIdx + 1) / questions.length) * 100}%`, background: `linear-gradient(90deg, ${theme.accent}, ${theme.accent2})`, borderRadius: 2, transition: "width 0.4s ease" }} />
        </div>
      </div>

      <div className="card">
        <div className="badge badge-blue" style={{ marginBottom: 12 }}>
          {q.type === "truefalse" ? "صح أو خطأ" : q.type === "mcq" ? "اختيار من متعدد" : "رتب الكلمات"}
        </div>
        <h3 style={{ fontFamily: G.fontDisplay, fontSize: 18, fontWeight: 700, marginBottom: 20, lineHeight: 1.5 }}>{q.question || (q.type === "order" && "رتب الكلمات لتكوّن جملة صحيحة")}</h3>

        {q.type === "truefalse" && (
          <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 10 }}>
            {[true, false].map(v => (
              <button key={String(v)} onClick={() => !confirmed && setSelected(v)}
                style={{ padding: "16px", borderRadius: 12, border: `2px solid ${selected === v ? (v ? theme.accent3 : theme.danger) : theme.border}`, background: selected === v ? (v ? `${theme.accent3}22` : `${theme.danger}22`) : theme.surface, color: theme.text, cursor: "pointer", fontWeight: 600, fontSize: 18, transition: "all 0.2s" }}>
                {v ? "✓ صح" : "✗ خطأ"}
              </button>
            ))}
          </div>
        )}

        {q.type === "mcq" && (
          <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
            {q.options.map((opt, i) => (
              <button key={i} onClick={() => !confirmed && setSelected(i)}
                style={{ padding: "12px 16px", borderRadius: 10, border: `2px solid ${selected === i ? theme.accent : theme.border}`, background: selected === i ? `${theme.accent}22` : theme.surface, color: theme.text, cursor: "pointer", textAlign: "left", transition: "all 0.2s", fontFamily: G.fontBody }}>
                <span style={{ fontFamily: G.fontMono, color: theme.accent, marginRight: 8 }}>{String.fromCharCode(65 + i)}.</span> {opt}
              </button>
            ))}
          </div>
        )}

        {q.type === "order" && (
          <div>
            <div style={{ display: "flex", flexWrap: "wrap", gap: 8, marginBottom: 16 }}>
              {orderWords.map((w, i) => (
                <button key={i} onClick={() => { if (!confirmed) { const arr = orderWords.filter((_, j) => j !== i); setOrderWords([...arr, w]); } }}
                  style={{ padding: "8px 14px", borderRadius: 8, background: theme.surface, border: `1px solid ${theme.border}`, color: theme.text, cursor: "pointer", fontFamily: G.fontMono }}>
                  {w}
                </button>
              ))}
            </div>
            <div style={{ minHeight: 44, padding: "10px 14px", borderRadius: 10, background: `${theme.accent}11`, border: `1px solid ${theme.accent}33`, fontFamily: G.fontMono, color: theme.accent }}>
              {orderWords.join(" ") || "اضغط على الكلمات لرتيبها"}
            </div>
          </div>
        )}

        {confirmed && (
          <div style={{ marginTop: 16, padding: 14, borderRadius: 10, background: isCorrect() ? `${theme.accent3}22` : `${theme.danger}22`, border: `1px solid ${isCorrect() ? theme.accent3 : theme.danger}33` }}>
            <div style={{ fontWeight: 700, color: isCorrect() ? theme.accent3 : theme.danger, marginBottom: 4 }}>
              {isCorrect() ? "✓ إجابة صحيحة!" : "✗ إجابة خاطئة"}
            </div>
            {!isCorrect() && q.type !== "order" && (
              <div style={{ fontSize: 13, color: theme.muted }}>
                الإجابة الصحيحة: {q.type === "truefalse" ? (q.answer ? "صح" : "خطأ") : q.options[q.answer]}
              </div>
            )}
          </div>
        )}
      </div>

      {!confirmed ? (
        <button className="btn btn-primary btn-lg" style={{ width: "100%" }} disabled={selected === null && q.type !== "order"} onClick={confirm}>
          تأكيد الإجابة
        </button>
      ) : (
        <button className="btn btn-primary btn-lg" style={{ width: "100%" }} onClick={next}>
          {qIdx < questions.length - 1 ? "السؤال التالي" : "عرض النتيجة"} <Icon name="arrow_right" size={16} />
        </button>
      )}
    </div>
  );
}

      { icon: "3️⃣", text: "التطبيق جاهز!", sub: "يفتح بدون متصفح" },
    ],
  };

  const platform = isIOS ? "ios" : isAndroid ? "android" : "desktop";
  const platformLabel = isIOS ? "iPhone / iPad" : isAndroid ? "Android" : "الكمبيوتر";

  return (
    <div style={{ position: "fixed", inset: 0, background: "#000a", backdropFilter: "blur(8px)", zIndex: 1000, display: "flex", alignItems: "flex-end", justifyContent: "center" }} onClick={onClose}>
      <div onClick={e => e.stopPropagation()} style={{ background: theme.card, borderRadius: "24px 24px 0 0", padding: 28, width: "100%", maxWidth: 480, border: `1px solid ${theme.border}`, borderBottom: "none", animation: "fadeUp 0.3s ease" }}>
        {/* Handle */}
        <div style={{ width: 40, height: 4, background: theme.border, borderRadius: 2, margin: "0 auto 24px" }} />

        <div style={{ display: "flex", alignItems: "center", gap: 14, marginBottom: 24 }}>
          <div style={{ width: 52, height: 52, borderRadius: 16, background: `linear-gradient(135deg, ${theme.accent}, ${theme.accent2})`, display: "flex", alignItems: "center", justifyContent: "center", boxShadow: `0 0 24px ${theme.accent}44` }}>
            <Icon name="zap" size={24} />
          </div>
          <div>
            <h2 style={{ fontFamily: G.fontDisplay, fontSize: 20, fontWeight: 800 }}>تثبيت التطبيق</h2>
            <p style={{ fontSize: 13, color: theme.muted, marginTop: 2 }}>Smart English Builder — {platformLabel}</p>
          </div>
        </div>

        {/* Tabs */}
        {deferredPrompt && (
          <div style={{ display: "flex", gap: 8, marginBottom: 20 }}>
            {["auto", "manual"].map(t => (
              <button key={t} onClick={() => setTab(t)} className={`btn ${tab === t ? "btn-primary" : "btn-ghost"}`} style={{ flex: 1, fontSize: 13 }}>
                {t === "auto" ? "⚡ تثبيت تلقائي" : "📖 دليل يدوي"}
              </button>
            ))}
          </div>
        )}

        {/* Auto install */}
        {(tab === "auto" && deferredPrompt) && (
          <div style={{ display: "flex", flexDirection: "column", gap: 12 }}>
            <div className="card" style={{ background: `${theme.accent}11`, borderColor: `${theme.accent}33`, textAlign: "center", padding: 24 }}>
              <div style={{ fontSize: 40, marginBottom: 12 }}>📲</div>
              <p style={{ color: theme.muted, fontSize: 14, lineHeight: 1.6 }}>اضغط الزر أدناه لتثبيت التطبيق مباشرة على جهازك بدون متجر تطبيقات!</p>
            </div>
            {installed ? (
              <div style={{ textAlign: "center", padding: 20, color: theme.accent3, fontWeight: 700, fontSize: 18 }}>✓ تم التثبيت بنجاح!</div>
            ) : (
              <button className="btn btn-primary btn-lg" style={{ width: "100%", animation: "glow 3s ease infinite" }} onClick={handleInstall}>
                <Icon name="plus" size={18} /> ثبّت التطبيق الآن
              </button>
            )}
          </div>
        )}

        {/* Manual steps */}
        {(tab === "manual" || !deferredPrompt) && (
          <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
            <p style={{ fontSize: 13, color: theme.muted, marginBottom: 4 }}>اتبع هذه الخطوات لتثبيت التطبيق على {platformLabel}:</p>
            {steps[platform].map((s, i) => (
              <div key={i} style={{ display: "flex", gap: 14, alignItems: "flex-start", padding: "12px 14px", background: theme.surface, borderRadius: 12, border: `1px solid ${theme.border}` }}>
                <span style={{ fontSize: 22, flexShrink: 0 }}>{s.icon}</span>
                <div>
                  <div style={{ fontWeight: 600, fontSize: 14 }}>{s.text}</div>
                  <div style={{ fontSize: 12, color: theme.muted, marginTop: 2 }}>{s.sub}</div>
                </div>
              </div>
            ))}

            {/* Platform switcher */}
            <div style={{ marginTop: 8 }}>
              <p style={{ fontSize: 12, color: theme.muted, marginBottom: 8 }}>جهاز مختلف؟</p>
              <div style={{ display: "flex", gap: 6 }}>
                {[["ios","🍎 iPhone"],["android","🤖 Android"],["desktop","💻 كمبيوتر"]].map(([p, l]) => (
                  <button key={p} onClick={() => {}} className="btn btn-ghost" style={{ flex: 1, fontSize: 11, padding: "8px 6px", opacity: platform === p ? 1 : 0.5 }}>{l}</button>
                ))}
              </div>
            </div>
          </div>
        )}

        <button className="btn btn-ghost" style={{ width: "100%", marginTop: 16 }} onClick={onClose}>إغلاق</button>
      </div>
    </div>
  );
}

// ─── BOTTOM NAV ──────────────────────────────────────────────────────────────
function BottomNav({ active, onNav }) {
  const tabs = [
    { id: "dashboard", icon: "home", label: "الرئيسية" },
    { id: "add", icon: "plus", label: "إضافة" },
    { id: "review", icon: "refresh", label: "مراجعة" },
    { id: "history", icon: "book", label: "السجل" },
    { id: "settings", icon: "settings", label: "إعدادات" },
  ];
  return (
    <div style={{ position: "fixed", bottom: 0, left: 0, right: 0, background: `${theme.surface}ee`, backdropFilter: "blur(20px)", borderTop: `1px solid ${theme.border}`, display: "flex", zIndex: 100, maxWidth: 480, margin: "0 auto" }}>
      {tabs.map(t => (
        <button key={t.id} onClick={() => onNav(t.id)}
          style={{ flex: 1, display: "flex", flexDirection: "column", alignItems: "center", gap: 4, padding: "10px 0 14px", background: "none", border: "none", cursor: "pointer", color: active === t.id ? theme.accent : theme.muted, transition: "color 0.2s" }}>
          <div style={{ transform: active === t.id ? "scale(1.15)" : "scale(1)", transition: "transform 0.2s" }}>
            <Icon name={t.icon} size={20} />
          </div>
          <span style={{ fontSize: 10, fontFamily: G.fontBody, fontWeight: active === t.id ? 600 : 400 }}>{t.label}</span>
        </button>
      ))}
    </div>
  );
}

// ─── APP ──────────────────────────────────────────────────────────────────────
export default function App() {
  const [user, setUser] = useState(() => storage.get("currentUser"));
  const [screen, setScreen] = useState("dashboard");
  const [words, setWords] = useState(() => storage.get("words") || []);
  const [settings, setSettings] = useState(() => storage.get("settings") || { lang: "ar", daily: 5 });
  const [todayReviewed, setTodayReviewed] = useState(() => storage.get("reviewedDate") === today());

  const [showInstall, setShowInstall] = useState(false);

  const saveWords = (newWords) => {
    const updated = [...words, ...newWords];
    setWords(updated);
    storage.set("words", updated);
    setScreen("dashboard");
  };

  const completeReview = () => {
    setTodayReviewed(true);
    storage.set("reviewedDate", today());
    setScreen("add");
  };

  const handleLogin = (u) => {
    setUser(u);
    storage.set("currentUser", u);
  };

  const handleLogout = () => {
    setUser(null);
    storage.set("currentUser", null);
    setScreen("dashboard");
  };

  const handleSettings = (s) => {
    setSettings(s);
    storage.set("settings", s);
  };

  if (!user) return (
    <>
      <style>{globalCSS}</style>
      <WelcomeScreen onLogin={handleLogin} />
    </>
  );

  return (
    <>
      <style>{globalCSS}</style>
      <div style={{ maxWidth: 480, margin: "0 auto", minHeight: "100vh", position: "relative" }}>
        {screen === "dashboard" && <Dashboard user={user} words={words} todayReviewed={todayReviewed} onNav={setScreen} onInstall={() => setShowInstall(true)} />}
        {screen === "add" && <AddWords words={words} todayReviewed={todayReviewed} onSave={saveWords} onNav={setScreen} />}
        {screen === "review" && <ReviewScreen words={words} onComplete={completeReview} />}
        {screen === "history" && <HistoryScreen words={words} />}
        {screen === "settings" && <SettingsScreen settings={settings} onSave={handleSettings} user={user} onLogout={handleLogout} />}
        <BottomNav active={screen} onNav={setScreen} />
        {showInstall && <InstallModal onClose={() => setShowInstall(false)} />}
      </div>
    </>
  );
}
